MACRO 16777220 "Экспорт 1" INV_Scroll_01

/run u="player"n=UnitName(u)_,c=UnitClass(u)l=UnitLevel(u)b,z,v=UnitAttackPower(u)h=UnitHealthMax(u)t={}for i=1,3 do _,_,q=GetTalentTabInfo(i) t[i]=q end print("WEXP1|"..n.."|"..c.."|"..l.."|"..b+z+v.."|"..h.."|"..table.concat(t,"/"))

END
